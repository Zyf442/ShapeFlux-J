      }
    }
  }

  private static double eval(double[] x, double[] bias, int i, int c) {
    double v0 = x[i];
    double v1 = 1.25;
    double v2 = v0 * v1;
    double v3 = bias[c];
    double v4 = v2 + v3;
    double v5 = Math.max(0.0, v4);
    double v6 = 0.5;
    double v7 = v5 * v6;
    double v8 = Math.tanh(v7);
    double v9 = 0.1;
    double v10 = v5 * v9;
    double v11 = v8 + v10;
    double v12 = io.shapeflux.ir.Ops.sigmoid(v11);
    return v12;
  }
}
